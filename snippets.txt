deleteCard(id: ID!): DeleteCardOutput!   

type DeleteCardOutput {     
  cardID: ID
}

      deleteCard: (_: unknown, { id }: { id: string }, { db }: Ctx) => {
        const kid = toInt(id);
        const existing = db
          .prepare('SELECT id, text, "order", column_id FROM cards WHERE id = ?')
          .get(kid);
        if (!existing) return { cardID: null };

        // Re-pack the source column's order to fill the gap.
        tx(db, () => {
          db.prepare("DELETE FROM cards WHERE id = ?").run(kid);
          db.prepare(
            'UPDATE cards SET "order" = "order" - 1 WHERE column_id = ? AND "order" > ?'
          ).run(existing.column_id, existing.order);
        });

        return { cardID: toID(kid) };
      },

const [, deleteCard] = useMutation(
  graphql(`
    mutation deleteCard($cardID: ID!) {
      deleteCard(id: $cardID) {
        cardID @Card_delete
      }
    }
  `),
)

deleteCard({
  variables: { cardID: card.id },
  optimisticResponse: {
    deleteCard: {
      cardID: card.id,
    },
  },
})
